    74→    74→    }
    75→    75→    else {

    90→    90→        }
    91→    91→    }

    93→    93→        pres = reflexive_pronouns[sujeto] + " " + pres;
    94→    94→    }
    95→    95→    spanish.push(pres);
    96→    96→
    97→    97→    //imperfect

   100→   100→        imp = attr.specialForms["imp"][sujeto];

   105→   105→            imp = reflexive_pronouns[sujeto] +" "+ imp;
   106→   106→        }
   107→   107→    }
   108→   108→    spanish.push(imp);
   109→   109→
   110→   110→    //pret
   111→   111→    var pret;

   113→   113→        pret = getEllos(stem, attr, sujeto);
   114→   114→        if(attr.isReflexive) {
   115→   115→            pret = "se " + pret;
   116→   116→        }
   117→   117→    } else {
   118→   118→        if(attr.specialForms["pret"] == null) {
   119→   119→            if(sujeto == YO && attr.endingType == AR) {
   120→   120→                if(stem.substring(stem.length - 1) == "z") {
   121→   121→                    pret = stem.substring(0, stem.length - 1) + "c" + pret_endings[suj_idx][sujeto];
   122→   122→                } else if(stem.substring(stem.length - 1) == "g") {
   123→   123→                    pret = stem.substring(0, stem.length - 1) + "gu" + pret_endings[suj_idx][sujeto];
   124→   124→                } else if(stem.substring(stem.length - 1) == "c") {
   125→   125→                    pret = stem.substring(0, stem.length - 1) + "qu" + pret_endings[suj_idx][sujeto];
   126→   126→                } else {
   127→   127→                    pret = stem + pret_endings[suj_idx][sujeto];
   128→   128→                }
   129→   129→            } else if(suj_idx == 1 && (stem.substring(stem.length - 1) == "e") || stem.substring(stem.length - 1) == "a") {
   130→   130→                var ending = pret_endings[suj_idx][sujeto];
   131→   131→                ending = "í" + ending.substring(1);
   132→   132→                pret = stem + ending;
   133→   133→            } else {
   134→   134→                pret = stem + pret_endings[suj_idx][sujeto];
   135→   135→            }
   136→   136→        } else {
   137→   137→            pret = attr.specialForms["pret"][sujeto];
   138→   138→        }
   139→   139→        if(attr.isReflexive) {
   140→   140→            pret = reflexive_pronouns[sujeto] +" "+ pret;
   141→   141→        }
   142→   142→    }
   143→   143→    spanish.push(pret);
   144→   144→
   145→   145→    //present perfect
   146→   146→    if(attr.isReflexive) {
   147→   147→        spanish.push(part.replace("-", haber_pres[sujeto]));
   148→   148→    } else {
   149→   149→        spanish.push(haber_pres[sujeto] + " " + part);
   150→   150→    }
   151→   151→
   152→   152→    //pluperfect
   153→   153→    if(attr.isReflexive) {
   154→   154→        spanish.push(part.replace("-", "hab" + imp_endings[1][sujeto]));
   155→   155→    } else {
   156→   156→        spanish.push("hab" + imp_endings[1][sujeto] + " " + part);
   157→   157→    }
   158→   158→
   159→   159→    //future
   160→   160→    var fut;
   161→   161→    if(attr.specialForms["future"] == null) {
   162→   162→        fut = plain_verb + fut_endings[sujeto];
   163→   163→    } else {
   164→   164→        fut = attr.specialForms["future"] + fut_endings[sujeto];
   165→   165→    }
   166→   166→    if(attr.isReflexive) {
   167→   167→        fut = reflexive_pronouns[sujeto] + " " + fut;
   168→   168→    }
   169→   169→    spanish.push(fut);
   170→   170→
   171→   171→    //fut perf
   172→   172→    if(attr.isReflexive) {
   173→   173→        spanish.push(part.replace("-", "habr" + fut_endings[sujeto]));
   174→   174→    } else {
   175→   175→        spanish.push("habr" + fut_endings[sujeto] + " " + part);
   176→   176→    }
   177→   177→
   178→   178→    //cond
   179→   179→    var cond;
   180→   180→    if(attr.specialForms["future"] == null) {
   181→   181→        cond = plain_verb + imp_endings[1][sujeto];
   182→   182→    } else {
   183→   183→        cond = attr.specialForms["future"] + imp_endings[1][sujeto];
   184→   184→    }
   185→   185→    if(attr.isReflexive) {
   186→   186→        cond = reflexive_pronouns[sujeto] + " " + cond;
   187→   187→    }
   188→   188→    spanish.push(cond);
   189→   189→
   190→   190→    //cond perf
   191→   191→    if(attr.isReflexive) {
   192→   192→        spanish.push(part.replace("-", "habr" + imp_endings[1][sujeto]));
   193→   193→    } else {
   194→   194→        spanish.push("habr" + imp_endings[1][sujeto] + " " + part);
   195→   195→    }
   196→   196→
   197→   197→    //pres subj
   198→   198→    spanish.push(getPresSubj(stem, attr, sujeto));
   199→   199→
   200→   200→    //imp subj
   201→   201→    var imp_subj = getEllos(stem, attr, ELLOS);
   202→   202→    imp_subj = imp_subj.substring(0, imp_subj.length - 3);
   203→   203→    if(attr.isReflexive) {
   204→   204→        imp_subj = reflexive_pronouns[sujeto] + " " + imp_subj;
   205→   205→    }

   207→   207→        imp_subj = addAccent(imp_subj + " ").substring(0, imp_subj.length);
   208→   208→    }

   210→   210→
   211→   211→    //pres. perf subj
   212→   212→    if(attr.isReflexive) {
   213→   213→        spanish.push(part.replace("-", haber_subj[sujeto]));
   214→   214→    } else {
   215→   215→        spanish.push(haber_subj[sujeto] + " " + part);
   216→   216→    }
   217→   217→
   218→   218→    //pluperf subj
   219→   219→    if(attr.isReflexive) {
   220→   220→        spanish.push([part.replace("-", haber_imp_subj[0][sujeto]), part.replace("-", haber_imp_subj[1][sujeto])]);
   221→   221→    } else {
   222→   222→        spanish.push([haber_imp_subj[0][sujeto] + " " + part, haber_imp_subj[1][sujeto] + " " + part]);

   225→   225→    // Tu +

   228→   228→        tup = attr.specialForms["command"];

   231→   231→        tup = attr.specialForms["pres"][2];
   232→   232→    }

   234→   234→        if(attr.specialForms["boot"] == null) {
   235→   235→            tup = stemChange(stem, attr.stemChangeType, false);    
   236→   236→        } else {
   237→   237→            tup = attr.specialForms["boot"];
   238→   238→        }
   239→   239→        if(attr.isReflexive) {
   240→   240→            tup = addAccent(tup);
   241→   241→            tup += pres_endings[attr.endingType][2] + reflexive_pronouns[1];
   242→   242→        } else {
   243→   243→            tup += pres_endings[attr.endingType][2];
   244→   244→        }
   245→   245→    }
   246→   246→    spanish.push([tup, "¡" + tup + "!"]);
   247→   247→
   248→   248→    //tun

   250→   250→
   251→   251→    //ud

   254→   254→        var temp = base_ud.substring(base_ud.length - 1);
   255→   255→        base_ud = base_ud.substring(3, base_ud.length - 1);
   256→   256→        base_ud = addAccent(base_ud) + temp + "se";
   257→   257→    }
   258→   258→    spanish.push([base_ud, "¡" + base_ud + "!"]);
   259→   259→
   260→   260→    //uds
   261→   261→    var base_uds = getPresSubj(stem, attr, 4);
   262→   262→    if(attr.isReflexive) {
   263→   263→        var temp = base_uds.substring(base_uds.length - 2);
   264→   264→        base_uds = base_uds.substring(3, base_uds.length - 2);
   265→   265→        base_uds = addAccent(base_uds) + temp + "se";
   266→   266→    }
   267→   267→    spanish.push([base_uds, "¡" + base_uds + "!"]);
   268→   268→
   269→   269→    //vosotros
   270→   270→    var vosn = "no " + getPresSubj(stem, attr, 5);

   272→   272→    if(attr.isReflexive) {
   273→   273→        if(attr.endingType == IR) {
   274→   274→            vosp = vosp.substring(0, vosp.length - 2) + "íos";
   275→   275→        } else {
   276→   276→            vosp = vosp.substring(0, vosp.length - 1) + "os";
   277→   277→        }
   278→   278→    }
   279→   279→    spanish.push([vosn, vosp, "¡" + vosn + "!", "¡" + vosp + "!"]);
   280→   280→    spanish.push([vosp, vosn, "¡" + vosp + "!", "¡" + vosn + "!"]);
   281→   281→
   282→   282→    //nosotros
   283→   283→    var nosotros;
   284→   284→    if(attr.specialForms["nos"] != null) {
   285→   285→        nosotros = attr.specialForms["nos"];
   286→   286→    } else {
   287→   287→        nosotros = getPresSubj(stem, attr, 3);
   288→   288→        if(attr.isReflexive) {
   289→   289→            nosotros = nosotros.substring(4, nosotros.length - 3);
   290→   290→            if(nosotros.substring(nosotros.length - 1) == "a") {
   291→   291→                nosotros = nosotros.substring(0, nosotros.length - 1) + "ámonos";
   292→   292→            } else if(nosotros.substring(nosotros.length - 1) == "e") {
   293→   293→                nosotros = nosotros.substring(0, nosotros.length - 1) + "émonos";
   294→   294→            }
   295→   295→        }
   296→   296→    }
   297→   297→    spanish.push([nosotros, "¡" + nosotros + "!"]);
   298→   298→    
   299→   299→    for(var i = 0; i < spanish.length; i++) {
   300→   300→        var obj = spanish[i];
   301→   301→        if(!(obj instanceof Array)) {
   302→   302→            obj = [obj];
   303→   303→        }
   304→   304→        if(attr.specialForms["participle"] != null && attr.specialForms["participle"] instanceof Array) {
   305→   305→            for(var j = 0; j < obj.length; j++) {
   306→   306→                if(obj[j].includes(attr.specialForms["participle"][0])) {
   307→   307→                    obj.push(obj[j].replace(attr.specialForms["participle"][0], attr.specialForms["participle"][1]));
   308→   308→                }
   309→   309→            }
   310→   310→        }
   311→   311→        spanish[i] = obj;
   312→   312→    }
   313→   313→
   314→   314→    //Then construct the english

   321→   321→        var be = false;
   322→   322→        var go = false;
   323→   323→        var have = false;
   324→   324→        var space = false;
   325→   325→        if(curr_ops[0] == "be worth" || curr_ops[0] == "be") {

   396→   396→        if(!be) {

   464→   464→// Get the most similar answer from the list given the (erroneous) input
   465→   465→function getMostSimilar(input, answers) {
   466→   466→    var minDist = 999;
   467→   467→    var minIdx = -1;
   468→   468→    var minLength = 999;
   469→   469→    for(var i = 0; i < answers.length; i++) {
   470→   470→        var curr_ans;
   471→   471→        if(answers[i] instanceof RegExp) {
   472→   472→            curr_ans = cleanString(answers[i].source);
   473→   473→        } else  {
   474→   474→            curr_ans = answers[i];
   475→   475→        }
   476→   476→        if(curr_ans.length < minLength) {
   477→   477→            minLength = curr_ans.length;
   478→   478→        }
   479→   479→        var sim_score = similarity(curr_ans, input);
   480→   480→        if(sim_score < minDist) {
   481→   481→            minDist = sim_score;
   482→   482→            minIdx = i;
   483→   483→        }
   484→   484→    }
   485→   485→    if(minDist == minLength)  {
   486→   486→        return answers[0] instanceof RegExp? cleanString(answers[0].source) : answers[0];
   487→   487→    } else {
   488→   488→        return answers[minIdx] instanceof RegExp? cleanString(answers[minIdx].source) : answers[minIdx];
   489→   489→    }
   490→   490→}

   497→   497→// Return how many changes are needed to make the strings the same (Levenshtein distance)
   498→   498→function similarity(str1, str2) {
   499→   499→    
   500→   500→    let dp = [];
   501→   501→    for (let i = 0; i <= str1.length; i++) {
   502→   502→        dp[i] = [i];
   503→   503→    }
   504→   504→    for (let j = 0; j <= str2.length; j++) {
   505→   505→        dp[0][j] = j;
   506→   506→    }
   507→   507→
   508→   508→    
   509→   509→    for (let i = 1; i <= str1.length; i++) {
   510→   510→        for (let j = 1; j <= str2.length; j++) {
   511→   511→            let cost = (str1.charAt(i - 1) == str2.charAt(j - 1)) ? 0 : 1;
   512→   512→            dp[i][j] = Math.min(
   513→   513→                dp[i - 1][j] + 1,     
   514→   514→                dp[i][j - 1] + 1,     
   515→   515→                dp[i - 1][j - 1] + cost  
   516→   516→            );
   517→   517→        }
   518→   518→    }
   519→   519→    
   520→   520→    return dp[str1.length][str2.length];
   521→   521→}

   668→   668→// Stem change the given verb stem
   669→   669→function stemChange(stem, type, oneLetter) {

   671→   671→        return stem;
   672→   672→    }
   673→   673→    var tbr;
   674→   674→    var i;
   675→   675→    for(i = stem.length - 1; i >= 0; i--) {

   678→   678→                if(oneLetter) {
   679→   679→                    tbr = stem.substring(0,i) + "i";
   680→   680→                    break;
   681→   681→                } else {
   682→   682→                    tbr = stem.substring(0,i) + "ie";
   683→   683→                    break
   684→   684→                }
   685→   685→            } else if(type == E_I) {
   686→   686→                tbr = stem.substring(0,i) + "i";
   687→   687→                break;
   688→   688→            }
   689→   689→        } else if(stem.substring(i, i+1) == "o") {
   690→   690→            if(type == O_UE) {
   691→   691→                if(oneLetter) {
   692→   692→                    tbr = stem.substring(0,i) + "u";
   693→   693→                    break;
   694→   694→                } else {
   695→   695→                    tbr = stem.substring(0,i) + "ue";
   696→   696→                    break;
   697→   697→                }
   698→   698→            }
   699→   699→        }
   700→   700→    }
   701→   701→    if(i != stem.length-1) {
